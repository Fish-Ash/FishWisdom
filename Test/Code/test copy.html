class DataProcessor:
    """处理和分析数据集的类"""
    
    def __init__(self, data):
        self.data = data
        self.processed = False
    
    def process_data(self):
        """清洗和预处理数据"""
        if not self.data:
            raise ValueError("数据不能为空")
        
        # 移除缺失值
        self.data = [item for item in self.data if item is not None]
        
        # 标准化数据
        self.normalize_data()
        
        self.processed = True
        return self.data
    
    def normalize_data(self):
        """将数据标准化到0-1范围"""
        if not self.data:
            return
        
        min_val = min(self.data)
        max_val = max(self.data)
        
        if max_val == min_val:
            self.data = [0.5] * len(self.data)
        else:
            self.data = [(x - min_val) / (max_val - min_val) 
                          for x in self.data]
