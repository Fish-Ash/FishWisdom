热力图

数据准备 

            import numpy as np

            x = np.linspace(-10, 10, 100)
            y = np.linspace(-10, 10, 100)

            X, Y = np.meshgrid(x, y)
            Z = X**2 + Y**2
            
            # Z是一个二维数组，表示每个点的值.
绘制热力图 

            import matplotlib.pyplot as plt

            plt.figure(figsize=(6, 5))
            plt.imshow(
                Z,  # Z是一个二维数组，表示每个点的值.
                extent=(x.min(), x.max(), y.min(), y.max()), # 指定坐标轴范围
                origin='lower', # 指定原点位置，'lower'表示原点在左下角 
                cmap='viridis'  # 指定颜色映射，'viridis'是一种常用的颜色映射方案
            )
            plt.colorbar(label='Z value')
            plt.title('Heatmap of Z = X^2 + Y^2')
            plt.xlabel('X-axis')
            plt.ylabel('Y-axis')
            plt.show()
