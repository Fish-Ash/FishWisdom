Matplotlib基础

创建画布 

            fig = plt.figure()
            """
            - figsize=(width, height)：指定画布的大小，单位为英寸.
            """

        

            fig, ax = plt.subplots(1,1,figsize=(6,4))
            # fig就是画布，就是窗口
            # ax就是子图，是画布上的一个区域，在这个区域上可以进行绘图
            # figsize是画布的大小，单位是英寸

            fig, ax = plt.subplots(2,2,figsize=(10,8))
            # 创建一个2行2列的画布，得到4个坐标轴对象
            # ax是一个2行2列的二维数组，可以通过ax[i,j]来访问第i行第j列的坐标轴对象
留出画布空白 

            plt.subplots_adjust()
            # 可供选择的参数：left, right, top, bottom
            # left can not be >= right
            # bottom can not be >= top
创建滑块条 

            from matplotlib.widgets import Slider

            # 创建滑块条
            ax_slider = plt.axes([0.25, 0.1, 0.65, 0.03]) # [left, bottom, width, height]
            slider = Slider(ax_slider, 'SliderName', 0.1, 10.0, valinit=1.0)

            # 定义滑块条的更新函数
            def update(val):
                # 在这里更新图表的内容
                fig.canvas.draw_idle() # 刷新图表

            slider.on_changed(update) # 将更新函数绑定到滑块条上
