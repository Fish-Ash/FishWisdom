工具使用

向模型声明函数工具

声明模板


            tools = [
                {
                    "type": "function", # 此处恒为function
                    "function": {
                        # 此处包含function的具体信息
                    }
                }
            ]

function信息介绍


            "function": {
                "name": ..., # 工具名称，不必须与函数同名
                "description": ..., # 必须，向模型解释工具作用，决定了模型在何时调用此工具（非常重要）
                "parameters":{
                    "type": "object", # parameters类型恒为object
                    "properties":{
                        # 工具中包含的参数
                    },
                    "required": [] # 工具所必需的参数，其中元素为properties名称
                }
            }

properties示例


            "properties": {
                "file_path": {
                    "type": "string",              # 参数类型
                    "description": "PDF文件的路径" # 参数描述
                }
            },
